// H12 one-off proof. Run only in the coordinator's admitted secretless namespace.
// No content is written below the Gateway; the child fixture provisions an EMPTY store.

const HEAD = "2f3f470f17424a40e86f8beaafd06b422dec8d2c";
const here = path.dirname(fileURLToPath(import.meta.url));
const repo = await fs.realpath(process.cwd());
const sha = (bytes) => createHash("sha256").update(bytes).digest("hex");
const load = (p) => import(pathToFileURL(path.join(repo, p)).href);
const emit = (kind, value) => console.log("H12 " + JSON.stringify({ kind, ...value }));
let stage = "preflight";
class ProofStop extends Error {
  constructor(code) { super(code); this.code = code; }
}
// Only literal categories leave this observer; no message/stack/cause serialization.
let primaryFailure;
const errorCodes = new Set(["ENOENT", "EACCES", "EPERM", "EADDRINUSE", "EADDRNOTAVAIL", "ESRCH", "EPIPE",
  "ECONNRESET", "ECONNREFUSED", "ETIMEDOUT", "ABORT_ERR", "ERR_SERVER_NOT_RUNNING", "ERR_MODULE_NOT_FOUND",
  "ERR_UNKNOWN_FILE_EXTENSION", "ERR_INVALID_ARG_TYPE", "ERR_INVALID_ARG_VALUE", "ERR_SPAWN_BROKER_UNAVAILABLE",
  "EPROCESSGROUP_CLEANUP_FAILED", "EPROCESS_TREE_VERIFICATION_UNSUPPORTED"]);
function errorKind(e) {
  if (e instanceof ProofStop) return "proof-constraint";
  if (errorCodes.has(e?.code)) return e.code;
  if (e?.name === "GatewayAgentAdmissionError") return "gateway-agent-admission-refused";
  if (e instanceof AggregateError) return "aggregate-error";
  if (e instanceof TypeError) return "type-error";
  if (e instanceof ReferenceError) return "reference-error";
  if (e instanceof SyntaxError) return "syntax-error";
  if (e instanceof RangeError) return "range-error";
  return "owner-operation-failed";
}
function safeError(e) {
  return { category: errorKind(e),
    ...(e instanceof ProofStop ? { check: e.code.split(":")[0] } : {}),
    ...(e instanceof AggregateError ? { members: e.errors.slice(0, 4).map(errorKind),
      omittedMembers: Math.max(0, e.errors.length - 4) } : {}) };
}
function rememberFailure(e, at = stage) {
  if (!primaryFailure) { primaryFailure = { stage: at, ...safeError(e) }; emit("primary-failure", primaryFailure); }
}
function exitFacts(p) {
  return { pidObserved: Number.isSafeInteger(p?.pid) && p.pid > 0,
    code: Number.isSafeInteger(p?.exitCode) ? p.exitCode : null,
    signal: p?.signalCode == null ? null :
      ["SIGTERM", "SIGKILL", "SIGINT", "SIGHUP", "SIGABRT", "SIGSEGV", "SIGBUS"].includes(p.signalCode) ? p.signalCode : "other" };
}
function requireFact(value, code) { if (!value) throw new ProofStop(code); }
function same(actual, expected, code) {
  requireFact(JSON.stringify(actual) === JSON.stringify(expected), code);
}
function inside(parent, child) {
  const rel = path.relative(parent, child);
  return rel !== "" && !rel.startsWith(".." + path.sep) && rel !== ".." && !path.isAbsolute(rel);
}
async function absent(filename) {
  try { await fs.lstat(filename); return false; }
  catch (e) { if (e.code === "ENOENT") return true; throw e; }
}
async function bindSources() {
  const manifest = JSON.parse(await fs.readFile(path.join(here, "source-bindings.json"), "utf8"));
  requireFact(manifest.head === HEAD, "manifest-head");
  for (const [p, bytes, hash, oid] of manifest.files) {
    const data = await fs.readFile(path.join(repo, p));
    const blob = createHash("sha1").update("blob " + data.length + "\0").update(data).digest("hex");
    requireFact(data.length === bytes && sha(data) === hash && blob === oid, "source-binding:" + p);
  }
  return manifest.files.length;
}

async function emptyFixture() {
  stage = "empty-registry-fixture";
  const root = await fs.realpath(process.env.H12_ROOT);
  const { readConfigFileSnapshot, resolveStateDir } = await load("src/config/config.ts");
  const { listAgentIds } = await load("src/plugin-sdk/agent-scope-runtime.ts");
  const { parseMembership, readStoresSection, effectiveBinding } = await load("extensions/knowledge/src/stores-config.ts");
  const { RegistryClient } = await load("extensions/knowledge/src/registry-client.ts");
  const snapshot = await readConfigFileSnapshot();
  requireFact(snapshot.exists && snapshot.valid, "fixture-config-invalid");
  const cfg = snapshot.config;
  // Require the canonical roster, not an implicit-agent fallback.
  requireFact(cfg.agents && (cfg.agents.entries || cfg.agents.list), "fixture-roster-absent");
  const agents = listAgentIds(cfg);
  requireFact(agents.length === 1 && /^[a-z][a-z0-9_-]{0,63}$/.test(agents[0]), "fixture-agent-ambiguous");
  const agent = agents[0];
  const membership = parseMembership(readStoresSection(cfg));
  requireFact(membership.issues.length === 0 && membership.invalid.size === 0, "fixture-membership-invalid");
  const binding = effectiveBinding(membership, agent);
  requireFact(binding === "agent:" + agent, "fixture-not-private-binding");
  const stateDir = resolveStateDir();
  requireFact(stateDir === process.env.OPENCLAW_STATE_DIR && inside(root, stateDir), "fixture-state-root");
  const knowledgeRoot = path.join(stateDir, "knowledge");
  requireFact(await absent(knowledgeRoot), "fixture-not-new");
  const registry = new RegistryClient(() => knowledgeRoot);
  let ready;
  try {
    const before = await registry.load();
    requireFact(Object.keys(before.stores).length === 0 && Object.keys(before.bindings).length === 0,
      "fixture-registry-not-empty");
    requireFact(before.pending === null && before.reconciled === null && Object.keys(before.grants).length === 0,
      "fixture-registry-preexisting-authority");
    await registry.call("provision", binding);
    const after = await registry.load();
    requireFact(after.pending === null && Object.keys(after.grants).length === 0, "fixture-unexpected-authority");
    requireFact(Object.keys(after.stores).length === 1 && Object.keys(after.bindings).length === 1,
      "fixture-store-count");
    const vaultId = after.bindings[binding];
    const store = after.stores[vaultId];
    requireFact(store && store.vaultId === vaultId && store.state === "ready" && store.present,
      "fixture-store-not-ready");
    requireFact(inside(knowledgeRoot, store.root) && inside(knowledgeRoot, store.statePath), "fixture-store-path");
    ready = { agent, binding, vaultId, root: store.root, statePath: store.statePath, publicationId: store.publicationId };
  } finally {
    // All owner operations above are sequential and settled before termination.
    await registry.dispose();
  }
  // Filesystem inspection is after the real registry worker's exit; no SQLite writes/reads here.
  const manifest = JSON.parse(await fs.readFile(path.join(ready.root, "vault.json"), "utf8"));
  requireFact(manifest.format === "knowledge-vault-v2" && manifest.generation === 0,
    "fixture-manifest-not-empty");
  requireFact(manifest.vaultId === ready.vaultId && manifest.publicationId === ready.publicationId,
    "fixture-manifest-identity");
  same((await fs.readdir(path.join(ready.root, "private"))).sort(), [], "fixture-content-present");
  requireFact(await absent(path.join(ready.root, "ontology.yaml")), "fixture-ontology-present");
  await fs.writeFile(path.join(root, "empty-store-receipt.json"), JSON.stringify(ready) + "\n", { flag: "wx", mode: 0o600 });
  emit("empty-store", { agent: ready.agent, vaultId: ready.vaultId, generation: 0,
    registryWorkerExited: true, contentRecords: 0, grantEntries: 0 });
}

async function gatewayProof() {
  requireFact(process.env.VITEST && process.env.OPENCLAW_TEST_RUNTIME_LOG === "1", "required-runtime-flags-missing");

  const root = await fs.mkdtemp(path.join(parent, "h12-gateway-"));

  const stateDir = path.join(root, "state");

    await fs.mkdir(path.isAbsolute(p) ? p : path.join(root, p), { recursive: true, mode: 0o700 });

    OPENCLAW_STATE_DIR: stateDir, OPENCLAW_CONFIG_PATH: configPath, OPENCLAW_PROFILE: profile,
    TMPDIR: path.join(root, "tmp"), XDG_CONFIG_HOME: path.join(root, "xdg-config"),
    XDG_CACHE_HOME: path.join(root, "xdg-cache"), XDG_DATA_HOME: path.join(root, "xdg-data"),
    XDG_RUNTIME_DIR: path.join(root, "xdg-runtime"), H12_ROOT: root,
    OPENCLAW_GATEWAY_TOKEN: randomUUID() + randomUUID() };
  const { runManagedCommand, hasUnjoinedWork } = await load("scripts/lib/managed-child-process.mts");
  const { createOpenClawTestProcess } = await load("test/helpers/openclaw-test-process.ts");
  const { testing } = await load("test/helpers/openclaw-test-instance.ts");
  const { reserveTestPortListener } = await load("src/test-utils/port-claims.ts");
  const { Check } = await import("typebox/value");
  const { contract } = await load("extensions/knowledge/src/contract.ts");
  const { BUILD_STAMP_FILE, RUNTIME_POSTBUILD_STAMP_FILE } = await load("scripts/lib/local-build-metadata-paths.mts");
  const cliEntry = path.join(repo, "dist", "index.js");
  const buildFacts = [];
  for (const p of [cliEntry, path.join(repo, "dist", BUILD_STAMP_FILE), path.join(repo, "dist", RUNTIME_POSTBUILD_STAMP_FILE)]) {
    const bytes = await fs.readFile(p);
    if (p !== cliEntry) {
      const stamp = JSON.parse(bytes.toString("utf8"));
      requireFact(stamp.head === HEAD && stamp.inputsClean === true, "unqualified-build-stamp");
    }
    buildFacts.push({ file: path.relative(repo, p), bytes: bytes.length, sha256: sha(bytes) });
  }
  // No call to the helper's auto-build, writeConfig, minimal Gateway or --allow-unconfigured path.
  const reservation = await reserveTestPortListener({ offsets: [0, 1], createListener: () => net.createServer(s => s.destroy()) });
  const port = reservation.claim.port;
  env.OPENCLAW_GATEWAY_PORT = String(port);
  const cliPrefix = [cliEntry, "--profile", profile];
  let reservationReleased = false, gateway, spawnCleanup, unsafeCommand = false;
  let gatewayExit, closeObserved = false, exitObserved = false, processError = false;
  let launchAttempted = false, startedResolved = false, spawnState = "not-published", spawnFailure;
  let bodyError;
  const custody = () => ({ launchAttempted, handleCreated: Boolean(gateway), startedResolved,
    exitObserved, closeObserved, spawnState, ...(spawnFailure ? { spawnFailure } : {}), ...exitFacts(gateway?.process) });
  let complete = false, cleanupOK = false;
  const requests = [];
  const stats = { commands: 0, stderrBytes: 0, gatewayStdoutBytes: 0, gatewayStderrBytes: 0 };
  async function command(args, label, timeoutMs = 60000) {
    stage = label;
    let output = [], bytes = 0, overflow = false, stderrBytes = 0, code;
    const started = Date.now();
    try {
      code = await runManagedCommand({ bin: process.execPath, args, cwd: repo, env,
        stdio: ["ignore", "pipe", "pipe"], timeoutMs, requireProcessTreeExit: true,
        onReady(child) {
          child.stdout.on("data", chunk => {
            bytes += chunk.length;
            if (bytes <= 2 * 1024 * 1024) output.push(Buffer.from(chunk)); else overflow = true;
          });
          child.stderr.on("data", chunk => { stderrBytes += chunk.length; });
        } });
    } catch (e) {
      unsafeCommand ||= hasUnjoinedWork(e);
      rememberFailure(e);
      emit("command-failed", { label, ...safeError(e), custodyUnsettled: unsafeCommand,
        elapsedMs: Date.now() - started, stdoutBytes: bytes, stderrBytes });
      // Do not echo error objects, argv, Gateway logs or authentication material.
      throw new ProofStop(unsafeCommand ? "command-custody-unsettled" : "command-execution-refused-or-timeout");
    } finally { stats.commands++; stats.stderrBytes += stderrBytes; }
    emit("command-terminal", { label, code, elapsedMs: Date.now() - started, stdoutBytes: bytes, stderrBytes });
    requireFact(!overflow, "command-output-bound");
    requireFact(code === 0, "command-nonzero:" + label);
    return Buffer.concat(output).toString("utf8");
  }
  async function cli(args, label) { return command([...cliPrefix, ...args], label); }
  async function cliJSON(args, label) {
    const raw = await cli(args, label);
    try { return JSON.parse(raw); } catch { throw new ProofStop("command-not-json:" + label); }
  }
  async function proxy(agent, operation, input, kind = "query") {
    const definition = contract.operations[operation];
    requireFact(definition && definition.kind === kind && Check(definition.input, input), "invalid-proof-input:" + operation);
    const payload = { pluginId: "knowledge", actionId: kind === "action" ? "agent_action" : "agent_query",
      payload: { agent, operation, input } };
    const answer = await cliJSON(["gateway", "call", "plugins.sessionAction", "--port", String(port),
      "--expect-url", "ws://127.0.0.1:" + port, "--params", JSON.stringify(payload), "--timeout", "30000", "--json"], operation);
    requireFact(answer && answer.ok === true, "operator-action-refused:" + operation);
    requireFact(Check(definition.output, answer.result), "invalid-proof-output:" + operation);
    return answer.result;
  }
  async function receipt(agent, input) {
    const frozen = JSON.stringify(input);
    requests.push(input);
    await fs.writeFile(path.join(root, "requests.json"), JSON.stringify(requests) + "\n", { mode: 0o600 });
    let value;
    try { value = await proxy(agent, "save_page", input, "action"); }
    catch (e) {
      rememberFailure(e);
      // Unknown outcome: inspect the SAME ID once; no resend. Hard operator refusal stops.
      if (!unsafeCommand && !(e instanceof ProofStop && e.code.startsWith("operator-action-refused:"))) {
        try {
          const observed = await proxy(agent, "mutation", { requestId: input.requestId, includeFiles: false });
          emit("uncertain-mutation", { requestId: input.requestId, payloadSha256: sha(frozen), receipt: observed.receipt });
        } catch { emit("uncertain-mutation", { requestId: input.requestId, payloadSha256: sha(frozen), reconciliation: "unavailable" }); }
      }
      throw e;
    }
    const settleUntil = Date.now() + 30000;
    while (value.state === "accepted" && Date.now() < settleUntil) {
      await delay(250);
      const observed = await proxy(agent, "mutation", { requestId: input.requestId, includeFiles: false });
      requireFact(observed.receipt, "accepted-receipt-missing");
      value = observed.receipt;
    }
    emit("publication", { payloadSha256: sha(frozen), receipt: value });
    requireFact(value.requestId === input.requestId && value.state === "published", "publication-not-published");
    requireFact(value.id && value.hash && value.revision >= 1 && value.indexed === false, "publication-incomplete");
    return value;
  }
  async function record(agent, saved, title, body) {
    const output = await proxy(agent, "record", { id: saved.id, metadataView: "full" });
    const r = output.record;
    requireFact(r && r.valid && r.kind === "page" && r.type === "note", "record-invalid");
    requireFact(r.id === saved.id && r.title === title && r.body === body && r.revision === saved.revision && r.hash === saved.hash,
      "record-content-identity");
    requireFact(output.content.offset === 0 && output.content.totalChars === body.length && output.content.nextOffset === null,
      "record-content-incomplete");
    const raw = await proxy(agent, "record", { id: saved.id, view: "raw", metadataView: "none" });
    requireFact(raw.record && typeof raw.record.raw === "string" && raw.content.nextOffset === null,
      "record-raw-incomplete");
    requireFact(sha(raw.record.raw) === saved.hash && raw.record.hash === saved.hash && raw.record.revision === saved.revision,
      "record-byte-hash");
    emit("record", { id: r.id, title, body, hash: r.hash, revision: r.revision, rawSha256: sha(raw.record.raw), valid: r.valid });
  }
  async function links(agent, id, direction, expected) {
    const output = await proxy(agent, "connections", { id, direction, limit: 100 });
    const ids = output.connections.map(c => c.record.id).sort();
    emit("connections", { id, direction, ids, total: output.total, nextOffset: output.nextOffset, snapshot: output.snapshot });
    same(ids, [...expected].sort(), "connections-membership:" + direction);
    requireFact(output.total === expected.length && output.nextOffset === null, "connections-completeness");
  }
  try {
    emit("begin", { head: HEAD, boundSourceFiles: count, buildFacts, versions: { node: process.versions.node, sqlite: process.versions.sqlite },
      proof: "empty-owner-fixture then real authenticated Gateway content operations", root });
    requireFact(await absent(configPath), "new-config-already-exists");

    same(await cliJSON(["config", "get", "meta.migrations.skillWorkshopPlugin", "--json"], "writer-metadata"), true, "writer-metadata-absent");
    const patch = { cron: { enabled: false }, gateway: { mode: "local", port, bind: "loopback", auth: { mode: "token" }, controlUi: { enabled: false } },
      plugins: { slots: { memory: "knowledge" }, entries: { knowledge: { enabled: true, config: { maintenance: { enabled: false } } },
        "skill-workshop": { config: { autonomous: { mode: "off" } } } } }, hooks: { enabled: false }, logging: { file: path.join(root, "logs", "gateway.log") } };
    const patchPath = path.join(root, "case.patch.json");
    await fs.writeFile(patchPath, JSON.stringify(patch) + "\n", { flag: "wx", mode: 0o600 });
    await cli(["config", "patch", "--file", patchPath], "canonical-config-patch");
    const validation = await cliJSON(["config", "validate", "--json"], "config-validate");

    });
    requireFact(!processError, "gateway-process-error");
    const stores = await cliJSON(["gateway", "call", "plugins.sessionAction", "--port", String(port), "--expect-url", "ws://127.0.0.1:" + port,
      "--params", JSON.stringify({ pluginId: "knowledge", actionId: "stores_snapshot", payload: { includeRecordCounts: false } }), "--timeout", "30000", "--json"], "stores-snapshot");
    requireFact(stores.ok === true && Check(contract.operations.stores_snapshot.output, stores.result), "stores-snapshot-refused");
    requireFact(stores.result.agents.some(a => a.id === agent && a.supported && a.binding === prepared.binding), "gateway-binding-mismatch");
    requireFact(stores.result.grants.entries.length === 0 && stores.result.stores.some(s => s.binding === prepared.binding && s.state === "ready"), "gateway-store-not-ready");

    emit("operator-preflight", { agent, vaultId: status.vault, store: status.store, records: status.records,
      writesEnabled: status.writesEnabled, ontologyFingerprint: ontology.fingerprint, grants: stores.result.grants.entries.length });
    const pages = [
      { title: "Jade Lantern", body: "A wholly fictional target page." },
      { title: "Copper Kite", body: "A wholly fictional unrelated page." },
      { title: "Visible navigation control", body: "[[Jade Lantern]]" },

    ];

    requireFact(new Set(saved.map(r => r.id)).size === 4, "page-identities-not-distinct");
    const bodies = [pages[3].body, "Fictional navigation: [[Jade Lantern]]", pages[3].body];
    for (let i = 0; i < bodies.length; i++) {
      if (i) {
        await record(agent, saved[3], pages[3].title, bodies[i - 1]);
        const before = saved[3];
        saved[3] = await receipt(agent, { requestId: randomUUID(), id: before.id, expectedRevision: before.revision,
          expectedHash: before.hash, title: pages[3].title, body: bodies[i], type: "note" });
        requireFact(saved[3].id === before.id && saved[3].revision === before.revision + 1 && saved[3].hash !== before.hash,
          "edit-revision-not-advanced");
      }
      for (let j = 0; j < pages.length; j++) await record(agent, saved[j], pages[j].title, j === 3 ? bodies[i] : pages[j].body);
      await links(agent, saved[3].id, "links", i === 1 ? [saved[0].id] : []);
      await links(agent, saved[0].id, "backlinks", i === 1 ? [saved[2].id, saved[3].id] : [saved[2].id]);
      await links(agent, saved[2].id, "links", [saved[0].id]);

    }
    const finalStatus = await proxy(agent, "status", {});
    requireFact(finalStatus.records === 4 && finalStatus.validRecords === 4 && finalStatus.vault === prepared.vaultId, "final-record-count");
    complete = true;
  } catch (e) {
    bodyError = e; rememberFailure(e);
    emit("failure-custody", custody());
  } finally {
    // Preserve body stage/error; each cleanup owner settles independently.
    let normalExit = false, stopped = false, listenerClosed = false, claimReleased = false;
    let cleanupFailed = false;
    async function clean(step, action) {
      emit("cleanup-step", { step, state: "started" });
      try { await action(); emit("cleanup-step", { step, state: "fulfilled" }); }
      catch (e) {
        cleanupFailed = true; rememberFailure(e, "cleanup:" + step);
        emit("cleanup-step", { step, state: "rejected", ...safeError(e) });
      }
    }
    if (gateway) {
      await clean("gateway-drain", async () => {
        if (gateway.process.exitCode === null && gateway.process.signalCode === null) gateway.process.kill("SIGTERM");
        const timeout = new AbortController();
        try {
          const exit = gateway.process.exitCode !== null || gateway.process.signalCode !== null
            ? { code: gateway.process.exitCode, signal: gateway.process.signalCode }
            : await Promise.race([gatewayExit, delay(60000, null, { signal: timeout.signal })]);
          normalExit = Boolean(exit && exit.code === 0 && exit.signal === null);
        } finally { timeout.abort(); }
      });
      await clean("gateway-stop", async () => { stopped = await gateway.stop(Date.now() + 30000, () => {}); });
      // stop joins completion and extinction. A rejected completion need not emit close.
      emit("gateway-terminal", { normalExit, stopped, ...custody() });
    } else if (spawnCleanup) {
      await clean("construction-cleanup", async () => { await spawnCleanup; });
    }
    if (!reservationReleased && reservation.listener.listening) {
      await clean("reservation-listener", async () => { await reservation.releaseListener(); reservationReleased = true; });
    }
    const probe = net.createServer(s => s.destroy());
    await clean("listener-rebind", async () => {
      await new Promise((resolve, reject) => { probe.once("error", reject); probe.listen(port, "127.0.0.1", resolve); });
      listenerClosed = true;
    });
    if (probe.listening) await clean("probe-close", () => new Promise((resolve, reject) => probe.close(e => e ? reject(e) : resolve())));
    const custodySettled = !unsafeCommand && (!launchAttempted || (gateway ? stopped : spawnState === "fulfilled"));
    if (custodySettled && listenerClosed && !probe.listening && !reservation.listener.listening) {
      await clean("port-claim", async () => { await reservation.claim.release(); claimReleased = true; });
    } else emit("cleanup-step", { step: "port-claim", state: "retained-unsettled" });
    cleanupOK = !cleanupFailed && custodySettled && (!gateway || normalExit) && listenerClosed && claimReleased;
    emit("terminal", { complete, proofComplete: complete && cleanupOK && !primaryFailure,
      cleanupOK, listenerClosed, claimReleased, ...custody(), retainedCase: root, ...stats });
  }
  if (bodyError !== undefined) throw bodyError;
  requireFact(cleanupOK, "cleanup-not-qualified");
  requireFact(complete, "proof-incomplete");
}

try {
  if (process.argv[2] === "empty-store") { await bindSources(); await emptyFixture(); }
  else { requireFact(process.argv.length === 2, "unexpected-driver-arguments"); await gatewayProof(); }
} catch (e) {
  // No raw error/cause/stack, config, token, connection identity, or authentication log egress.
  rememberFailure(e);
  emit("stopped", { ...primaryFailure, proofComplete: false });
  process.exitCode = 1;
}
