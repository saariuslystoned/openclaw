class ProofStop extends Error {
  code: string;
  constructor(code: string) { super(code); this.code = code; }
}
// Only literal categories leave this observer; no message/stack/cause serialization.
let primaryFailure;
const errorCodes = new Set(["ENOENT", "EACCES", "EPERM", "EADDRINUSE", "EADDRNOTAVAIL", "ESRCH", "EPIPE",
  "ECONNRESET", "ECONNREFUSED", "ETIMEDOUT", "ABORT_ERR", "ERR_SERVER_NOT_RUNNING", "ERR_MODULE_NOT_FOUND",
  "ERR_UNKNOWN_FILE_EXTENSION", "ERR_INVALID_ARG_TYPE", "ERR_INVALID_ARG_VALUE", "ERR_SPAWN_BROKER_UNAVAILABLE",
  "EPROCESSGROUP_CLEANUP_FAILED", "EPROCESS_TREE_VERIFICATION_UNSUPPORTED"]);
function errorKind(e) {
  if (e instanceof ProofStop) return "proof-constraint";
  if (errorCodes.has(e?.code)) return e.code;
  if (e?.name === "GatewayAgentAdmissionError") return "gateway-agent-admission-refused";
  if (e instanceof AggregateError) return "aggregate-error";
  if (e instanceof TypeError) return "type-error";
  if (e instanceof ReferenceError) return "reference-error";
  if (e instanceof SyntaxError) return "syntax-error";
  if (e instanceof RangeError) return "range-error";
  return "owner-operation-failed";
}
function safeError(e) {
  return { category: errorKind(e),
    ...(e instanceof ProofStop ? { check: e.code.split(":")[0] } : {}),
    ...(e instanceof AggregateError ? { members: e.errors.slice(0, 4).map(errorKind),
      omittedMembers: Math.max(0, e.errors.length - 4) } : {}) };
}
function rememberFailure(e, at = stage) {
  if (!primaryFailure) { primaryFailure = { stage: at, ...safeError(e) }; emit("primary-failure", primaryFailure); }
}
function exitFacts(p) {
  return { pidObserved: Number.isSafeInteger(p?.pid) && p.pid > 0,
    code: Number.isSafeInteger(p?.exitCode) ? p.exitCode : null,
    signal: p?.signalCode == null ? null :
      ["SIGTERM", "SIGKILL", "SIGINT", "SIGHUP", "SIGABRT", "SIGSEGV", "SIGBUS"].includes(p.signalCode) ? p.signalCode : "other" };
}
function requireFact(value, code) { if (!value) throw new ProofStop(code); }
function same(actual, expected, code) {
  requireFact(JSON.stringify(actual) === JSON.stringify(expected), code);
}
function inside(parent, child) {
  const rel = path.relative(parent, child);
  return rel !== "" && !rel.startsWith(".." + path.sep) && rel !== ".." && !path.isAbsolute(rel);
}
async function absent(filename) {
  try { await fs.lstat(filename); return false; }
  catch (e) { if (e.code === "ENOENT") return true; throw e; }
}
async function bindSources() {
  const manifestBytes = await fs.readFile(path.join(companion, "source-bindings.json"));
  requireFact(sha(manifestBytes) === "eebfe50244d575febc6c5dae8973f8112b57a9fec0a61f194167ad251ec8cb8c", "companion-manifest-hash");
  const manifest = JSON.parse(manifestBytes.toString("utf8"));
  requireFact(manifest.head === HEAD, "manifest-head");
  for (const [p, bytes, hash, oid] of manifest.files) {
    const data = await fs.readFile(path.join(repo, p));
    const blob = createHash("sha1").update("blob " + data.length + "\0").update(data).digest("hex");
    requireFact(data.length === bytes && sha(data) === hash && blob === oid, "source-binding:" + p);
  }
  return manifest.files.length;
}

async function observeNormalRunnerOwner() {
  stage = "normal-runner-owner-preflight";
  requireFact(process.platform === "linux" && !process.versions.bun, "requires-qualified-linux-node");
  const declared = runtimeProcessEntrypoints.serviceChildGroupAnchor;
  const declarationUrl = new URL(declared.currentModuleUrl);
  const inherited = getInheritedNativeProcessOwner();
  const selectedUrl = inherited ? new URL(inherited) : resolveRuntimeProcessEntrypointUrl("serviceChildGroupAnchor");
  requireFact(declarationUrl.protocol === "file:" && selectedUrl.protocol === "file:", "owner-file-url-required");
  const declarationFile = fileURLToPath(declarationUrl);
  const selectedFile = fileURLToPath(selectedUrl);
  const distRoot = resolveRuntimeProcessDistRoot(declarationFile);
  if (distRoot === undefined) throw new ProofStop("declared-runtime-dist-not-observed");
  // The owning resolver handles both flat emitted chunks and nested declarations.
  const declaredDist = path.resolve(distRoot);
  const generation = path.dirname(declaredDist);
  const generations = path.join(repo, ".artifacts/vitest-workers");
  const generationObserved = path.basename(declaredDist) === "dist" &&
    path.dirname(generation) === generations && path.basename(generation).startsWith("run-");
  const inGeneration = generationObserved && inside(declaredDist, selectedFile);
  const workerArgv = resolveRuntimeWorkerArgv(selectedUrl, process.execPath);
  const loaderFree = workerArgv.length === 1 && workerArgv[0] === selectedFile && /\.m?js$/.test(selectedFile);
  emit("owner-selection", {
    inherited: inherited !== undefined, generationObserved, selectedInGeneration: inGeneration, loaderFree,
    declarationRelativeUrl: inside(repo, declarationFile) ? path.relative(repo, declarationFile) : "outside-repository",
    selectedRelativeUrl: inside(repo, selectedFile) ? path.relative(repo, selectedFile) : "outside-repository",
    argvShape: workerArgv.map(a => a === selectedFile ? "entry" : a === "--import" ? "--import" : "other"),
  });
  requireFact(generationObserved && inGeneration && loaderFree, "normal-compiled-owner-not-observed");
  // This getter is optional in a pool process: bootstrap installs the descriptor
  // in the config process. The transformed declaration is the actual supplied path.
  const descriptor = getVitestWorkerDescriptor();
  requireFact(!descriptor || path.resolve(descriptor.directory) === generation, "runner-descriptor-mismatch");
  const manifestBytes = await fs.readFile(path.join(generation, "manifest.json"));
  const manifest = JSON.parse(manifestBytes.toString("utf8"));
  const selectedBytes = await fs.readFile(selectedFile);
  const declarationBytes = await fs.readFile(declarationFile);
  const selectedHash = sha(selectedBytes), declarationHash = sha(declarationBytes);
  const selectedKey = path.relative(declaredDist, selectedFile).replaceAll(path.sep, "/");
  const declarationKey = path.relative(declaredDist, declarationFile).replaceAll(path.sep, "/");
  requireFact(manifest.outputs?.[selectedKey] === selectedHash &&
    manifest.outputs?.[declarationKey] === declarationHash, "observed-owner-manifest-mismatch");
  requireFact(typeof manifest.identity === "string" && /^[a-f0-9]{64}$/.test(manifest.identity) &&
    Number.isFinite(manifest.durationMs) && manifest.durationMs >= 0, "runner-manifest-incomplete");
  emit("owner-preparation", { generation: path.relative(repo, generation), descriptorObserved: Boolean(descriptor),
    manifestSha256: sha(manifestBytes), identity: manifest.identity, preparationDurationMs: manifest.durationMs,
    inputCount: Object.keys(manifest.inputs).length, outputCount: Object.keys(manifest.outputs).length,
    cacheSignaturePresent: typeof manifest.cacheSignature === "string",
    selectedOwnerSha256: selectedHash, declarationSha256: declarationHash, selectedOwnerBytes: selectedBytes.length,
    verification: "two observed outputs match owner manifest; full verification and borrower join belong to runner" });
}

    const frozen = JSON.stringify(input);
    requests.push(input);

        try {
          const observed = await proxy(agent, "mutation", { requestId: input.requestId, includeFiles: false });
          emit("uncertain-mutation", { requestId: input.requestId, payloadSha256: sha(frozen), receipt: observed.receipt });
        } catch { emit("uncertain-mutation", { requestId: input.requestId, payloadSha256: sha(frozen), reconciliation: "unavailable" }); }
      }
      throw e;
    }
    const settleUntil = Date.now() + 30000;
    while (value.state === "accepted" && Date.now() < settleUntil) {
      await delay(250);
      const observed = await proxy(agent, "mutation", { requestId: input.requestId, includeFiles: false });
      requireFact(observed.receipt, "accepted-receipt-missing");
      value = observed.receipt;

    emit("publication", { payloadSha256: sha(frozen), receipt: value });
    requireFact(value.requestId === input.requestId && value.state === "published", "publication-not-published");
    requireFact(value.id && value.hash && value.revision >= 1 && value.indexed === false, "publication-incomplete");
    return value;
