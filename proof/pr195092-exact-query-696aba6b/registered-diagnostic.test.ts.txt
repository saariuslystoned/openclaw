import { createHash, randomUUID } from "node:crypto";
import { mkdir, readFile, writeFile } from "node:fs/promises";
import path from "node:path";
import type { FeatureInput } from "openclaw/plugin-sdk/feature-contract";
import type { OpenClawPluginToolFactory } from "openclaw/plugin-sdk/plugin-entry";
import { Check } from "typebox/value";
import { expect, it } from "vitest";
import { stringify } from "yaml";
import { contract } from "./contract.js";
import plugin from "./index.js";
import { isFullHit } from "./retrieval-hit.js";
import { afterTest, sessionActions, storeFixture, toolContext, useTestTempDirs } from "./test-support.js";

// Proof-only. Run unchanged once per phase in the coordinator's isolated namespace.
// The existing fixture mints SYNTHETIC host/tool authority; this is not Gateway proof.
const dirs = useTestTempDirs();
const sha = (bytes: string | Buffer) => createHash("sha256").update(bytes).digest("hex");
type Phase = "base" | "external";
type Mode = "all" | "phrase" | "any";
type Query = "mixed" | "nfc" | "nfd";
const queries = {
  mixed: "\u03ac\u03bb\u03c6\u03b1 \u03b2\u03b7\u0301\u03c4\u03b1",
  nfc: "\u03ac\u03bb\u03c6\u03b1 \u03b2\u03ae\u03c4\u03b1",
  nfd: "\u03b1\u0301\u03bb\u03c6\u03b1 \u03b2\u03b7\u0301\u03c4\u03b1",
} as const;
// Already reported P2: PR195092 issuecomment-6718823955; not a new finding.
const korean = "\uac00\u1102\u1161";
const rows = [
  { id: "mixedform_nfc", title: "Fictional NFC row", body: queries.nfc },
  { id: "mixedform_nfd", title: "Fictional NFD row", body: queries.nfd },
  { id: "mixedform_mixed", title: "Fictional mixed row", body: queries.mixed },
  { id: "mixedform_ascii", title: "Fictional ASCII control", body: "cobalt orchard" },
  { id: "mixedform_korean", title: "Fictional Korean literal", body: korean },
] as const;

// Independently literal predicted sets, not generated by the production expression.
// These are diagnostic predictions, NOT the desired canonical-equivalence contract.
const expected: Record<Phase, Record<Mode, Record<Query, readonly string[]>>> = {
  base: {
    all: { mixed: ["mixedform_mixed"], nfc: ["mixedform_nfc"], nfd: ["mixedform_nfd"] },
    phrase: { mixed: ["mixedform_mixed"], nfc: ["mixedform_nfc"], nfd: ["mixedform_nfd"] },
    any: {
      mixed: ["mixedform_mixed", "mixedform_nfc", "mixedform_nfd"],
      nfc: ["mixedform_mixed", "mixedform_nfc"],
      nfd: ["mixedform_mixed", "mixedform_nfd"],
    },
  },
  external: {
    all: {
      mixed: ["mixedform_nfc", "mixedform_nfd"],
      nfc: ["mixedform_nfc", "mixedform_nfd"],
      nfd: ["mixedform_nfc", "mixedform_nfd"],
    },
    phrase: {
      mixed: ["mixedform_nfc", "mixedform_nfd"],
      nfc: ["mixedform_nfc", "mixedform_nfd"],
      nfd: ["mixedform_nfc", "mixedform_nfd"],
    },
    any: {
      mixed: ["mixedform_mixed", "mixedform_nfc", "mixedform_nfd"],
      nfc: ["mixedform_mixed", "mixedform_nfc", "mixedform_nfd"],
      nfd: ["mixedform_mixed", "mixedform_nfc", "mixedform_nfd"],
    },
  },
};
const moduleHashes: Record<Phase, string> = {
  base: "a558b8f30ea9f8e17a95d8477aa457ee51cac1624cb40fe434ae36dd2f285a9e",
  external: "98eebd73f7dce7ed7c9b4dce348885646c76376ba1b68f7a03f2383c964bec18",
};

it("observes PR195092 mixed-form exact-match retention through registered knowledge_grep", async () => {
  const phase = process.env.H21_MIXED_PHASE;
  if (phase !== "base" && phase !== "external") {
    throw new Error("Set H21_MIXED_PHASE=base|external in the qualified disposable proof namespace");
  }
  const moduleBytes = await readFile(new URL("./retrieval.ts", import.meta.url));
  expect(sha(moduleBytes), "full active retrieval.ts binding").toBe(moduleHashes[phase]);
  const scalars = Array.from(queries.mixed, (scalar) => scalar.codePointAt(0));
  expect(scalars, "actual mixed Unicode scalar sequence; no textual escape witness").toEqual([
    0x03ac, 0x03bb, 0x03c6, 0x03b1, 0x0020, 0x03b2, 0x03b7, 0x0301, 0x03c4, 0x03b1,
  ]);
  expect(queries.mixed.includes("\\")).toBe(false);
  const koreanScalars = Array.from(korean, (scalar) => scalar.codePointAt(0));
  expect(koreanScalars, "existing reviewer P2 actual scalar sequence").toEqual([0xac00, 0x1102, 0x1161]);
  expect(korean.includes("\\")).toBe(false);
  expect(queries.nfc.normalize("NFD")).toBe(queries.nfd);
  expect(queries.mixed.normalize("NFC")).toBe(queries.nfc);
  expect(queries.mixed.normalize("NFD")).toBe(queries.nfd);
  console.info("H21_MIXED_BEGIN", JSON.stringify({
    phase, versions: process.versions, moduleSha256: sha(moduleBytes),
    mixedScalars: scalars, mixedUtf8Hex: Buffer.from(queries.mixed).toString("hex"),
    koreanScalars, koreanUtf8Hex: Buffer.from(korean).toString("hex"),
    authority: "synthetic existing test-support host/toolContext; NOT Gateway authority",
  }));

  const services: { stop?: () => unknown }[] = [];
  const factories = new Map<string, OpenClawPluginToolFactory>();
  let stopped = false;
  async function stop() {
    const failures: unknown[] = [];
    for (const service of services.splice(0)) {
      try { await service.stop?.(); } catch (error) { failures.push(error); }
    }
    if (failures.length) throw failures[0];
    stopped = true;
  }
  // Fallback if setup fails; normal stop happens in finally before temp removal.
  afterTest(stop);
  const fixture = await storeFixture(dirs.make("h21-pr195092-mixed-"), {
    writesEnabled: false,
    policy: { writesEnabled: false },
    extra: {
      ...sessionActions(),
      registerService(service: { stop?: () => unknown }) { services.push(service); },
      registerTool(factory: OpenClawPluginToolFactory, options: { name: string }) {
        factories.set(options.name, factory);
      },
    },
  });
  const sourceFiles = new Map<string, { filename: string; bytes: Buffer; hash: string }>();
  for (const row of rows) {
    const filename = path.join(fixture.root, "private", "notes", row.id + ".md");
    const raw = "---\n" + stringify({
      format: "knowledge-record-v1", id: row.id, revision: 1, kind: "page", type: "note",
      fields: {}, title: row.title, scope: { visibility: "private" },
      recorded_at: "2026-09-23T00:00:00Z", updated_at: "2026-09-23T00:00:00Z",
      lifecycle: "active",
    }) + "---\n" + row.body;
    await mkdir(path.dirname(filename), { recursive: true });
    await writeFile(filename, raw, { flag: "wx" });
    sourceFiles.set(row.id, { filename, bytes: Buffer.from(raw), hash: sha(raw) });
  }
  plugin.register(fixture.api);
  const context = toolContext(fixture.api).tool;
  async function invoke(operation: "grep" | "record" | "lookup", input: Record<string, unknown>) {
    const tool = factories.get("knowledge_" + operation)?.(context);
    if (!tool || Array.isArray(tool)) throw new Error("Missing registered knowledge_" + operation);
    return (await tool.execute(randomUUID(), input)).details as unknown;
  }
  async function grep(input: FeatureInput<typeof contract, "grep">) {
    const output = await invoke("grep", input);
    console.info("H21_MIXED_QUERY", JSON.stringify({ phase, input, output }));
    if (!Check(contract.operations.grep.output, output)) throw new Error("Invalid registered grep output");
    expect(output.index.state).toBe("ready");
    expect(output.index.lag).toBe(false);
    expect(output.coverage.complete).toBe(true);
    expect(output.total).toBe(output.hits.length);
    expect(output.queryMode).toBe(input.mode);
    for (const hit of output.hits) {
      if (!isFullHit(hit)) throw new Error("Expected requested full hit");
      const row = rows.find((item) => item.id === hit.record.id);
      const source = sourceFiles.get(hit.record.id);
      if (!row || !source) throw new Error("Unexpected non-fixture result");
      expect(hit.record.valid).toBe(true);
      expect(hit.citation.hash).toBe(source.hash);
      expect(hit.citation.revision).toBe(1);
      expect(hit.snippet.field).toBe("body");
      expect(hit.snippet.text).toBe(row.body.slice(hit.snippet.start, hit.snippet.end));
    }
    return output.hits.map((hit) => hit.record.id).sort();
  }
  async function record(id: string) {
    const output = await invoke("record", { id });
    if (!Check(contract.operations.record.output, output) || !output.record) {
      throw new Error("Missing/invalid registered record output: " + id);
    }
    if (typeof output.record.body !== "string") throw new Error("Expected record content: " + id);
    return {
      body: output.record.body, hash: output.record.hash,
      revision: output.record.revision, valid: output.record.valid,
    };
  }
  async function exact(id: string) {
    const output = await invoke("lookup", { query: id, detail: "full" });
    console.info("H21_MIXED_EXACT", JSON.stringify({ phase, id, output }));
    if (!Check(contract.operations.lookup.output, output)) throw new Error("Invalid registered lookup output");
    expect(output.outcome).toBe("found");
    expect(output.hits.map((hit) => hit.record.id)).toEqual([id]);
  }
  const before = new Map<string, { body: string; hash: string; revision: number }>();
  const observations: { mode: Mode; query: Query; ids: string[] }[] = [];
  let collected = false;
  try {
    for (const row of rows) {
      const value = await record(row.id);
      expect(value.body).toBe(row.body);
      expect(value.hash).toBe(sourceFiles.get(row.id)!.hash);
      expect(value.revision).toBe(1);
      expect(value.valid).toBe(true);
      before.set(row.id, { body: value.body, hash: value.hash, revision: value.revision });
      await exact(row.id);
    }
    // Complete and print all nine actual mode/form observations before predictions.
    for (const mode of ["all", "phrase", "any"] as const) {
      for (const query of ["mixed", "nfc", "nfd"] as const) {
        observations.push({ mode, query, ids: await grep({ query: queries[query], mode, detail: "full" }) });
      }
    }
    // One additional exact query for the already published single-token P2.
    const koreanIds = await grep({ query: korean, mode: "all", detail: "full" });
    const ascii: { mode: Mode; ids: string[] }[] = [];
    for (const mode of ["all", "phrase", "any"] as const) {
      ascii.push({ mode, ids: await grep({ query: "cobalt orchard", mode, detail: "full" }) });
    }
    const unrelated = await grep({ query: "quartzmissingtoken", mode: "all", detail: "full" });
    for (const row of rows) {
      const value = await record(row.id);
      expect({ body: value.body, hash: value.hash, revision: value.revision }).toEqual(before.get(row.id));
      const source = sourceFiles.get(row.id)!;
      expect(await readFile(source.filename)).toEqual(source.bytes);
    }
    collected = true;
    console.info("H21_MIXED_OBSERVATIONS", JSON.stringify({
      phase, observations, koreanIds, ascii, unrelated,
      sourceHashes: Object.fromEntries([...sourceFiles].map(([id, source]) => [id, source.hash])),
      recordsUnchanged: true,
    }));
    for (const control of ascii) expect(control.ids).toEqual(["mixedform_ascii"]);
    expect(unrelated).toEqual([]);
    for (const observation of observations) {
      expect.soft(observation.ids, "prediction " + phase + "/" + observation.mode + "/" + observation.query)
        .toEqual(expected[phase][observation.mode][observation.query]);
    }
    expect.soft(koreanIds, "prediction " + phase + "/all/korean")
      .toEqual(phase === "base" ? ["mixedform_korean"] : []);
    expect.soft(koreanIds, "EXISTING_REVIEW_P2_KOREAN_LITERAL_RETENTION").toContain("mixedform_korean");
    // Separate true behavior invariant from predicted defect: a byte-identical mixed
    // query must retain its own row. External phase is predicted to FAIL both checks.
    // Such failure is witness evidence, never a passing-product/green classification.
    for (const mode of ["all", "phrase"] as const) {
      const observation = observations.find((item) => item.mode === mode && item.query === "mixed");
      if (!observation) throw new Error("Missing exact mixed observation");
      expect.soft(observation.ids, "EXACT_MIXED_RETENTION/" + mode).toContain("mixedform_mixed");
    }
  } finally {
    await stop();
    for (const source of sourceFiles.values()) {
      expect(await readFile(source.filename)).toEqual(source.bytes);
    }
    expect(sha(await readFile(new URL("./retrieval.ts", import.meta.url)))).toBe(moduleHashes[phase]);
    console.info("H21_MIXED_TERMINAL", JSON.stringify({
      phase, observationsCollected: collected, workersStopped: stopped,
      sourceFilesUnchanged: true, moduleUnchanged: true,
      meaning: "diagnostic observations and cleanup only; NOT product/Gateway/whole-PR clearance",
    }));
  }
});
