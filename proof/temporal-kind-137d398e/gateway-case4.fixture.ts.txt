import { createHash, randomUUID } from "node:crypto";
import fs from "node:fs/promises";
import net from "node:net";
import path from "node:path";
import { expect, it } from "vitest";
import { runManagedCommand } from "../../../scripts/lib/managed-child-process.mts";
import { createOpenClawTestState } from "../../../src/test-utils/openclaw-test-state.js";
import { reserveTestPortListener } from "../../../src/test-utils/port-claims.js";
import { testing as gatewayTesting } from "../../../test/helpers/openclaw-test-instance.js";
import { createOpenClawTestProcess } from "../../../test/helpers/openclaw-test-process.js";

it("observes temporal guidance through the canonically bootstrapped shipped Gateway", async () => {
  const profile = "h5-kind-proof-case4";
  const state = await createOpenClawTestState({ label: profile, scenario: "empty", applyEnv: false });
  const claim = await reserveTestPortListener({ offsets: [0, 1], createListener: () => net.createServer(socket => socket.destroy()) });
  const port = claim.claim.port;
  let listenerReleased = false;
  const releaseListener = async () => { if (!listenerReleased) { await claim.releaseListener(); listenerReleased = true; } };
  const opaque = `gateway-h5-kind-proof-case4-${randomUUID()}`;
  const env = { ...state.env, OPENCLAW_PROFILE: profile, OPENCLAW_TEST_RUNTIME_LOG: "1", OPENCLAW_GATEWAY_PORT: String(port), OPENCLAW_GATEWAY_TOKEN: opaque, OPENCLAW_TEST_MINIMAL_GATEWAY: "0", OPENCLAW_SKIP_CHANNELS: "1", OPENCLAW_SKIP_PROVIDERS: "1", OPENCLAW_SKIP_GMAIL_WATCHER: "1", OPENCLAW_SKIP_CRON: "1", OPENCLAW_SKIP_BROWSER_CONTROL_SERVER: "1", OPENCLAW_SKIP_CANVAS_HOST: "1" };
  const scrub = (value: string) => value.split(opaque).join("[REDACTED]").split("\n").filter(line => !/password|authorization|cookie|secret|OPENCLAW_GATEWAY_TOKEN/i.test(line)).join("\n");
  const commands: unknown[] = [];
  async function run(label: string, args: string[], timeoutMs = 45000) {
    let child: any; let stdout = ""; let stderr = "";
    await runManagedCommand({ bin: process.execPath, args, cwd: process.cwd(), env, stdio: ["ignore", "pipe", "pipe"], shell: false, timeoutMs, timeoutKillGraceMs: 0, onReady: proc => { child = proc; proc.stdout?.setEncoding("utf8"); proc.stderr?.setEncoding("utf8"); proc.stdout?.on("data", x => { stdout += x; }); proc.stderr?.on("data", x => { stderr += x; }); } });
    const publicArgs = args.map(x => x === opaque ? "[REDACTED]" : x === String(port) ? "[owned isolated port]" : x.startsWith("ws://") ? "[private loopback]" : x);
    commands.push({ label, argv: ["node", ...publicArgs], exitCode: child.exitCode, stdoutRedacted: scrub(stdout), stderrRedacted: scrub(stderr), stdoutSha256: createHash("sha256").update(stdout).digest("hex"), stderrSha256: createHash("sha256").update(scrub(stderr)).digest("hex") });
    if (child.exitCode !== 0) throw new Error(`${label}: CLI exit ${child.exitCode}: ${scrub(stderr + stdout)}`);
    return stdout;
  }
  const cli = (label: string, args: string[], timeout?: number) => run(label, ["dist/index.js", "--profile", profile, ...args], timeout);
  let gateway: Awaited<ReturnType<typeof createOpenClawTestProcess>> | undefined;
  let constructionCleanup: Promise<unknown> | undefined;
  const out: string[] = [], err: string[] = [], readiness: any[] = [];
  try {
    await expect(fs.access(state.configPath)).rejects.toThrow();
    const workspace = path.join(state.root, "workspace");
    const setup = JSON.parse((await cli("canonical baseline setup", ["setup", "--baseline", "--skip-bootstrap", "--workspace", workspace, "--json"], 60000)).trim());
    expect(setup).toMatchObject({ ok: true, configStatus: "created", configPath: state.configPath, workspaceDir: workspace });
    expect(JSON.parse((await cli("writer-owned plugin provenance", ["config", "get", "meta.migrations.skillWorkshopPlugin", "--json"])).trim())).toBe(true);
    const patch = await run("unchanged author patch producer", ["/proof/h5-case4-make-patch.mjs", state.root, String(port)]);
    const patchFile = path.join(state.root, "case4.patch.json"); await fs.writeFile(patchFile, patch);
    await cli("canonical config patch", ["config", "patch", "--file", patchFile]);
    const validation = await cli("canonical config validation", ["config", "validate", "--json"]);
    const slot = JSON.parse((await cli("selected memory slot", ["config", "get", "plugins.slots.memory", "--json"])).trim()); expect(slot).toBe("knowledge");
    expect(JSON.parse((await cli("disabled synthetic cron", ["config", "get", "cron.enabled", "--json"])).trim())).toBe(false);
    expect(JSON.parse((await cli("workshop autonomy off", ["config", "get", "plugins.entries.skill-workshop.config.autonomous.mode", "--json"])).trim())).toBe("off");
    await releaseListener();
    gateway = await createOpenClawTestProcess({ argv: [process.execPath, "dist/index.js", "--profile", profile, "gateway", "run", "--port", String(port), "--bind", "loopback"], cwd: process.cwd(), env, gracefulShutdownTimeoutMs: 5000, onError: e => err.push(scrub(e.message)), onSpawnCleanup: cleanup => { constructionCleanup = cleanup; } });
    gateway.process.stdout.on("data", x => out.push(String(x))); gateway.process.stderr.on("data", x => err.push(String(x)));
    await gateway.started;
    await gatewayTesting.waitForGatewayReady(gateway.process, out, err, port, 90000, fetch, undefined, d => readiness.push(d));
    expect(gateway.process.pid).toBeGreaterThan(0);
    for (const includePlugins of [true, false]) {
      const response = await cli(`observed catalog includePlugins=${includePlugins}`, ["gateway", "call", "tools.catalog", "--port", String(port), "--expect-url", `ws://127.0.0.1:${port}`, "--params", JSON.stringify({ includePlugins }), "--timeout", "30000", "--json"]);
      const observed = JSON.parse(response.trim()); await fs.writeFile(`/proof/h5-case4-catalog-${includePlugins ? "plugins" : "core"}.json`, JSON.stringify(observed, null, 2) + "\n");
    }
    const checked = await run("unchanged author offline response checker", ["/proof/h5-case4-check-catalog.mjs", "/proof/h5-case4-catalog-plugins.json", "/proof/h5-case4-catalog-core.json"]);
    await fs.writeFile("/proof/h5-case4-catalog-check.json", checked);
    const receipt = { head: "137d398ed80839087a44e1efd25eead82d517752", entrypoint: "dist/index.js", newState: true, baseline: { ok: setup.ok, configStatus: setup.configStatus }, writerOwnedProvenance: true, validation: JSON.parse(validation.trim()), memorySlot: slot, normalPluginStartup: true, gatewayPid: gateway.process.pid, readiness: readiness.map(d => ({ outcome: d.outcome, elapsedMs: d.elapsedMs, attempts: d.attempts })), scope: "actual built CLI/Gateway catalog; normal plugin startup; side-effect services skipped, no model calls or host/outbound network", commands };
    await fs.writeFile("/proof/h5-case4-observed-receipt.json", JSON.stringify(receipt, null, 2) + "\n"); console.info("H5_CASE2_SHIPPED_ENTRYPOINT", JSON.stringify(receipt));
  } catch (error) {
    await fs.writeFile("/proof/h5-case4-failed-command-receipt.json", JSON.stringify({ commands }, null, 2) + "\n");
    throw new Error(scrub(error instanceof Error ? error.message : String(error)));
  } finally {
    let processClosed = true;
    if (gateway) processClosed = await gateway.stop(Date.now() + 20000, message => err.push(scrub(message)));
    else if (constructionCleanup) await constructionCleanup;
    await releaseListener(); await claim.claim.release(); await state.cleanup();
    await fs.writeFile("/proof/h5-case4-cleanup.json", JSON.stringify({ processClosed, canonicalStateCleanup: true, logicalPortClaimReleased: true, stateAndOpaqueCredentialTmpfsOnly: true }) + "\n");
    expect(processClosed).toBe(true);
  }
}, 240000);
